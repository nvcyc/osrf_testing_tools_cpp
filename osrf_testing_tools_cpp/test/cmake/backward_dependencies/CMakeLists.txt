cmake_minimum_required(VERSION 3.20)
project(test_backward_dependencies LANGUAGES CXX)

# Isolate dependency discovery from the libraries available on the test host.
foreach(name LIBDW_LIBRARY LIBDW_INCLUDE_DIR LIBDWARF_LIBRARY LIBDWARF_INCLUDE_DIR)
  set(${name} "" CACHE STRING "" FORCE)
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/bfd.h" "")
set(LIBBFD_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/include" CACHE PATH "" FORCE)
if(CASE STREQUAL "shared")
  set(bfd_suffix "${CMAKE_SHARED_LIBRARY_SUFFIX}")
else()
  set(bfd_suffix "${CMAKE_STATIC_LIBRARY_SUFFIX}")
endif()
set(LIBBFD_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/libbfd${bfd_suffix}" CACHE FILEPATH "" FORCE)
file(WRITE "${LIBBFD_LIBRARY}" "")

foreach(name SFRAME ZSTD)
  if(CASE STREQUAL "without_optional_libraries")
    set(LIB${name}_LIBRARY "" CACHE FILEPATH "" FORCE)
  else()
    set(LIB${name}_LIBRARY "${CMAKE_CURRENT_BINARY_DIR}/lib${name}.a" CACHE FILEPATH "" FORCE)
    file(WRITE "${LIB${name}_LIBRARY}" "")
  endif()
endforeach()

include("${BACKWARD_CONFIG_FILE}")
get_target_property(libraries Backward::Backward INTERFACE_LINK_LIBRARIES)
if("${libraries}" MATCHES "NOTFOUND")
  message(FATAL_ERROR "Missing optional libraries leaked into the link interface: ${libraries}")
endif()
foreach(name SFRAME ZSTD)
  if(CASE STREQUAL "with_optional_libraries")
    if(NOT "${LIB${name}_LIBRARY}" IN_LIST libraries)
      message(FATAL_ERROR "Static BFD is missing ${name}: ${libraries}")
    endif()
  elseif(CASE STREQUAL "shared" AND "${LIB${name}_LIBRARY}" IN_LIST libraries)
    message(FATAL_ERROR "Shared BFD should not add static dependencies: ${libraries}")
  endif()
endforeach()
